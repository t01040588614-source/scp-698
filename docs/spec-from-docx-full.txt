// accounts.service.ts 내부에 추가export interface UpdateTradeDto {  side?: 'BUY' | 'SELL';  tradeDate?: string;  price?: number;  quantity?: number;  fee?: number | null;  tax?: number | null;  memo?: string | null;}@Injectable()export class AccountsService {  // ... 기존 생성자 및 메서드 ...  /**   * 매매 내역 수정   * - 내역 업데이트 후 해당 종목 포지션 재계산   */  async updateTrade(tradeId: number, dto: UpdateTradeDto) {    const trade = await this.tradeRepo.findOne({      where: { id: tradeId },      relations: ['account', 'stock'],    });    if (!trade) {      throw new NotFoundException('Trade not found');    }    if (dto.side != null) trade.side = dto.side;    if (dto.tradeDate != null) trade.tradeDate = new Date(dto.tradeDate);    if (dto.price != null) trade.price = dto.price;    if (dto.quantity != null) trade.quantity = dto.quantity;    if (dto.fee !== undefined) trade.fee = dto.fee ?? null;    if (dto.tax !== undefined) trade.tax = dto.tax ?? null;    if (dto.memo !== undefined) trade.memo = dto.memo ?? null;    trade.amount = (trade.price ?? 0) * (trade.quantity ?? 0);    await this.tradeRepo.save(trade);    // 포지션 재계산    await this.recalculatePosition(trade.account, trade.stock);    return trade;  }  /**   * 매매 내역 삭제   * - 삭제 후 해당 종목 포지션 재계산   */  async deleteTrade(tradeId: number) {    const trade = await this.tradeRepo.findOne({      where: { id: tradeId },      relations: ['account', 'stock'],    });    if (!trade) {      throw new NotFoundException('Trade not found');    }    const { account, stock } = trade;    await this.tradeRepo.remove(trade);    // 포지션 재계산    await this.recalculatePosition(account, stock);    return { ok: true };  }  /**   * 특정 매매 내역 상세   */  async getTrade(tradeId: number) {    const trade = await this.tradeRepo.findOne({      where: { id: tradeId },      relations: ['account', 'stock'],    });    if (!trade) {      throw new NotFoundException('Trade not found');    }    return trade;  }}

6. 이 프로젝트의 핵심 포인트 정리
단일 프로젝트 안에 다음이 모두 포함됨:
한국 주식 시장 데이터 대시보드 (Investing.com 스타일)
종목별 차트 + 뉴스 + 컨센서스
사용자별 계좌 관리 + 매매 내역 + 평가손익 자동 계산
일별 자산 추이 차트
매매 내역 수정/삭제 및 자동 포지션 재계산
확장 포인트:
증권사 API 연동 (자동 매매 내역 동기화)
인증/인가 (JWT, OAuth)
알림 (실적 발표, 컨센서스 변경, 목표가 상향 등)
백테스트/포트폴리오 최적화 등
