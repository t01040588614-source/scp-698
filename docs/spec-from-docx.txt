카서한테 보낼껀데 좀 더 디테일 적으로 알려줘
아래는 Cursor(Cursor IDE) 에게 “이 프로젝트의 전체 설계 + 구현 세부사항”을 한 번에 전달할 수 있도록, 매우 디테일한 명세서 + 코드 가이드 형태입니다. 이 내용을 그대로 Cursor 의 채팅/프롬프트로 넣으면, 프로젝트 생성부터 확장까지 일관된 맥락으로 작업할 수 있습니다.
Cursor 용 프로젝트 명세서 (한국 주식 대시보드 + 스크리너)
0. 프로젝트 목표
https://kr.investing.com/equities/south-korea 페이지를 참고하여,
한국 주식 종목 목록 + 필터/정렬 (스크리너)
종목 상세 (시세, 재무, 차트, 뉴스, 컨센서스)
사용자별 스크리너 저장 기능
을 제공하는 웹 애플리케이션을 구축한다.

4.2.2. stocks.service.ts (전체)
apps/api/src/stocks/stocks.service.ts
import { Injectable } from '@nestjs/common';import { InjectRepository } from '@nestjs/typeorm';import { Repository } from 'typeorm';import { Stock } from '../entities/stock.entity';import { StockQuote } from '../entities/stock-quote.entity';import { StockFundamental } from '../entities/stock-fundamental.entity';import { News } from '../entities/news.entity';import { Consensus } from '../entities/consensus.entity';import { UserScreener } from '../entities/user-screener.entity';export interface ListStocksDto {  query?: string;  market?: string;  sector?: string;  industry?: string;  marketCapGte?: number;  marketCapLte?: number;  perGte?: number;  perLte?: number;  pbrGte?: number;  pbrLte?: number;  dividendYieldGte?: number;  sortBy?: 'marketCap' | 'per' | 'pbr' | 'dividendYield' | 'name';  sortOrder?: 'ASC' | 'DESC';  limit?: number;  offset?: number;}@Injectable()export class StocksService {  constructor(    @InjectRepository(Stock)    private stockRepo: Repository<Stock>,    @InjectRepository(StockQuote)    private quoteRepo: Repository<StockQuote>,    @InjectRepository(StockFundamental)    private fundRepo: Repository<StockFundamental>,    @InjectRepository(News)    private newsRepo: Repository<News>,    @InjectRepository(Consensus)    private consensusRepo: Repository<Consensus>,    @InjectRepository(UserScreener)    private screenerRepo: Repository<UserScreener>,  ) {}  async listStocks(dto: ListStocksDto) {    const {      query,      market,      sector,      industry,      marketCapGte,      marketCapLte,      perGte,      perLte,      pbrGte,      pbrLte,      dividendYieldGte,      sortBy = 'marketCap',      sortOrder = 'DESC',      limit = 50,      offset = 0,    } = dto;    const qb = this.stockRepo      .createQueryBuilder('s')      .leftJoinAndSelect('s.fundamentals', 'f')      .orderBy(`s.${sortBy}`, sortOrder)      .limit(limit)      .offset(offset);    if (query) {      qb.andWhere('s.name LIKE :q OR s.code LIKE :q OR s.nameEn LIKE :q', {        q: `%${query}%`,      });    }    if (market) {      qb.andWhere('s.market = :market', { market });    }    if (sector) {      qb.andWhere('s.sector = :sector', { sector });    }    if (industry) {      qb.andWhere('s.industry = :industry', { industry });    }    if (marketCapGte != null) {      qb.andWhere('s.marketCap >= :marketCapGte', { marketCapGte });    }    if (marketCapLte != null) {      qb.andWhere('s.marketCap <= :marketCapLte', { marketCapLte });    }    if (perGte != null) {      qb.andWhere('f.per >= :perGte', { perGte });    }    if (perLte != null) {      qb.andWhere('f.per <= :perLte', { perLte });    }    if (pbrGte != null) {      qb.andWhere('f.pbr >= :pbrGte', { pbrGte });    }    if (pbrLte != null) {      qb.andWhere('f.pbr <= :pbrLte', { pbrLte });    }    if (dividendYieldGte != null) {      qb.andWhere('f.dividendYield >= :dividendYieldGte', {        dividendYieldGte,      });    }    const [stocks, total] = await qb.getManyAndCount();    return { stocks, total };  }  async getStockDetail(code: string) {    const stock = await this.stockRepo.findOne({      where: { code },      relations: ['fundamentals'],    });    if (!stock) {      return null;    }    const latestQuote = await this.quoteRepo.findOne({      where: { stock: { code } },      order: { date: 'DESC' },    });    return {      stock,      latestQuote,    };  }  async getStockQuotes(code: string, limit = 200) {    const quotes = await this.quoteRepo.find({      where: { stock: { code } },      order: { date: 'ASC' },      take: limit,      select: ['date', 'open', 'high', 'low', 'close', 'volume'],    });    return quotes.map((q) => ({      date: q.date.toISOString().slice(0, 10),      open: Number(q.open) ?? 0,      high: Number(q.high) ?? 0,      low: Number(q.low) ?? 0,      close: Number(q.close) ?? 0,      volume: Number(q.volume) ?? 0,    }));  }  async getStockNews(code: string, limit = 20) {    const stock = await this.stockRepo.findOne({ where: { code } });    if (!stock) return [];    const news = await this.newsRepo.find({      where: [{ stock: { code } }, { stock: null }],      order: { publishedAt: 'DESC' },      take: limit,    });    return news.map((n) => ({      id: n.id,      title: n.title,      summary: n.summary,      url: n.url,      source: n.source,      publishedAt: n.publishedAt?.toISOString(),      sentiment: n.sentiment,    }));  }  async getStockConsensus(code: string) {    const stock = await this.stockRepo.findOne({ where: { code } });    if (!stock) return null;    const consensus = await this.consensusRepo.findOne({      where: { stock: { code } },      order: { date: 'DESC' },    });    if (!consensus) return null;    return {      date: consensus.date.toISOString().slice(0, 10),      targetPriceAvg: consensus.targetPriceAvg,      targetPriceHigh: consensus.targetPriceHigh,      targetPriceLow: consensus.targetPriceLow,      buyCount: consensus.buyCount,      holdCount: consensus.holdCount,      sellCount: consensus.sellCount,    };  }  async saveScreener(name: string, conditions: any, createdBy?: string | null) {    const screener = this.screenerRepo.create({      name,      conditions,      createdBy: createdBy ?? null,    });    return this.screenerRepo.save(screener);  }  async listScreeners(createdBy?: string | null) {    const where: any = {};    if (createdBy) {      where.createdBy = createdBy;    }    return this.screenerRepo.find({      where,      order: { createdAt: 'DESC' },    });  }  async deleteScreener(id: number) {    await this.screenerRepo.delete(id);    return { ok: true };  }}

import { createAppDataSource } from './create-connection';import { StockQuote } from '../entities/stock-quote.entity';interface ExternalQuote {  code: string;  date: string;  open: number;  high: number;  low: number;  close: number;  volume: number;  amount?: number;}async function fetchExternalQuotes(  code: string,  from: string,  to: string,): Promise<ExternalQuote[]> {  // 실제 구현 시 일봉 API 호출  return [    {      code,      date: '2024-09-30',      open: 70000,      high: 71000,      low: 69500,      close: 70500,      volume: 10000000,      amount: 705000000000,    },  ];}export async function ingestQuotesForCode(  code: string,  from: string,  to: string,) {  const ds = await createAppDataSource();  try {    const quotes = await fetchExternalQuotes(code, from, to);    await ds.manager.upsert(      StockQuote,      quotes.map((q) => ({        stock: { code: q.code },        date: new Date(q.date),        open: q.open,        high: q.high,        low: q.low,        close: q.close,        volume: q.volume,        amount: q.amount ?? null,      })),      ['stock', 'date'],    );    console.log(`Quotes upserted for ${code}:`, quotes.length);  } finally {    await ds.destroy();  }}export async function ingestAllQuotes(from: string, to: string) {  const ds = await createAppDataSource();  try {    const stocks = await ds.manager.find(Stock, { select: ['code'] });    for (const s of stocks) {      await ingestQuotesForCode(s.code, from, to);    }  } finally {    await ds.destroy();  }}if (require.main === module) {  const to = new Date().toISOString().slice(0, 10);  const from = new Date(Date.now() - 30 * 86400000).toISOString().slice(0, 10);  ingestAllQuotes(from, to).catch(console.error);}
